cmake_minimum_required(VERSION 3.20)
project(script_registration_contract LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use the Win32 generator for the original VM ABI")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(SDGO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT EXISTS "${SDGO_LUA50_SOURCE}/src/lapi.c")
 message(FATAL_ERROR "Set SDGO_LUA50_SOURCE to existing official Lua 5.0.3 test sources")
endif()
file(GLOB LUA_CORE CONFIGURE_DEPENDS "${SDGO_LUA50_SOURCE}/src/*.c")
add_library(fixture_lua50 STATIC ${LUA_CORE}
 "${SDGO_LUA50_SOURCE}/src/lib/lauxlib.c"
 "${SDGO_LUA50_SOURCE}/src/lib/lbaselib.c"
 "${SDGO_LUA50_SOURCE}/src/lib/lstrlib.c")
target_include_directories(fixture_lua50 PUBLIC "${SDGO_LUA50_SOURCE}/include" PRIVATE "${SDGO_LUA50_SOURCE}/src")
target_compile_definitions(fixture_lua50 PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(script_registration STATIC
 "${SDGO_ROOT}/core/script_registration.cpp"
 "${SDGO_ROOT}/core/native_script_registration.cpp")
target_include_directories(script_registration PUBLIC "${SDGO_ROOT}/include")
target_compile_definitions(script_registration PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(script_registration PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(script_registration PUBLIC bcrypt)
add_executable(script_registration_test "${CMAKE_CURRENT_SOURCE_DIR}/../script_registration_test.cpp")
target_link_libraries(script_registration_test PRIVATE script_registration fixture_lua50)
target_compile_definitions(script_registration_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(script_registration_test PRIVATE /W4 /permissive- /utf-8)
target_link_options(script_registration_test PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO)
enable_testing()
add_test(NAME script_registration_real_lua50 COMMAND script_registration_test)
if(SDGO_SCRIPT_NATIVE_CLIENT)
 add_test(NAME script_registration_original_api_replay COMMAND script_registration_test --native-api "${SDGO_SCRIPT_NATIVE_CLIENT}")
endif()
